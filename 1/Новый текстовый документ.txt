    // Automatic weather cycle tied to day/night
    let weatherMode = 'none';
    
    function updateWeatherWithTime() {
      // Cycle: 0-10% night, 10-20% pre-dawn, 20-30% sunrise, 30-50% day, 50-60% sunset, 60-80% dusk, 80-100% night
      const now = new Date();
      const totalMS = 40000; // 40 second cycle
      const cycleMS = (now.getTime() % totalMS);
      const percent = (cycleMS / totalMS) * 100;
      
      let newWeather = 'none';
      
      // Sunrise (20-30%): light rain
      if (percent >= 20 && percent < 30) {
        newWeather = 'rain';
      }
      // Sunset (50-60%): heavier rain
      else if (percent >= 50 && percent < 60) {
        newWeather = 'rain';
      }
      // Night (80-100% and 0-10%): snow
      else if (percent >= 80 || percent < 10) {
        newWeather = 'snow';
      }
      
      // Update weather if it changed
      if (newWeather !== weatherMode) {
        weatherMode = newWeather;
        if (weatherMode === 'rain') {
          createRain();
          document.getElementById('snow-layer').innerHTML = '';
        } else if (weatherMode === 'snow') {
          createSnow();
          document.getElementById('rain-layer').innerHTML = '';
        } else {
          document.getElementById('rain-layer').innerHTML = '';
          document.getElementById('snow-layer').innerHTML = '';
        }
      }
      
      // Keep checking for weather updates
      requestAnimationFrame(updateWeatherWithTime);
    }
    
    // Start automatic weather cycle
